> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development. Steps use checkbox (`- [ ]`) syntax.

Detector consecutive_pair — дополнение в `anomalies/detector.ts`. Существующие invariants (line-coverage, byte-reconstruction) **обязаны держаться** на TRM.

## Контракты модулей

### `src/tools/_compact.ts`

```typescript
import type { TocNode } from "../index/types.js";

// CompactTocNode — TocNode shape with default-valued fields stripped.
// Returned by JSON.stringify-friendly shape for tool responses.
export type CompactTocNode = {
  id: string;
  level: 1 | 2 | 3 | 4 | 5 | 6;
  title: string;
  line: number;
  line_end: number;
  section_lines: number;
  numbering?: string;             // only if non-null
  is_likely_artifact?: true;      // only if true
  artifact_reason?: string;       // only if is_likely_artifact
  children?: CompactTocNode[];    // only if non-empty
};

export function compactTocNode(node: TocNode): CompactTocNode;
```

Behavior:
- Always include: `id`, `level`, `title`, `line`, `line_end`, `section_lines`.
- `numbering`: include only if `!== null`.
- `is_likely_artifact`: include only if `=== true` (never `false`).
- `artifact_reason`: include only if `is_likely_artifact === true` (т.е. имеется reason).
- `children`: include только если `children.length > 0`; рекурсивно через `compactTocNode`.

### `view_toc_response.ts` поведение

После всех существующих шагов (определение root subtree, cap-loop) сериализация использует `compactTocNode`. Response shape additive:

```typescript
type ViewTocResponse = {
  file: {...};
  toc: CompactTocNode[];          // вместо TocNode[]
  anomalies_summary: {...};
  truncated?: true;               // только при truncation
  effective_depth?: number;       // только при truncation
  hint?: string;                  // только при truncation
};
```

Алгоритм:
1. Resolve root subtree (full toc или children-of-start_id).
2. requested_depth = input.depth ?? 6.
3. for depth = requested_depth down to 1:
   - trimmed = trimToDepth(root, depth) → CompactTocNode[]
   - payload = JSON.stringify({ file, toc: trimmed, anomalies_summary })
   - if `Buffer.byteLength(payload) <= MAX_VIEW_TOC_BYTES`:
     - if `depth < requested_depth`: установить truncated=true, effective_depth=depth, hint=`"Tree trimmed from depth ${requested_depth} to ${depth}; use start_id=<a leaf id> to drill deeper."`
     - return
4. Если даже depth=1 не помещается:
   - take prefix of trimmed[] elements until cap; rest dropped.
   - truncated=true, effective_depth=1, hint=`"Document has too many root sections (${total}). Returned first ${returned}. Use start_id to navigate further."`
   - return

В `raw=true` mode: cap аналогично — отдаём prefix flat_headers до cap'а.

```typescript
// src/tools/_compact.ts
import type { TocNode } from "../index/types.js";

export type CompactTocNode = {
  id: string;
  level: 1 | 2 | 3 | 4 | 5 | 6;
  title: string;
  line: number;
  line_end: number;
  section_lines: number;
  numbering?: string;
  is_likely_artifact?: true;
  artifact_reason?: string;
  children?: CompactTocNode[];
};

export function compactTocNode(node: TocNode): CompactTocNode {
  const result: CompactTocNode = {
    id: node.id,
    level: node.level,
    title: node.title,
    line: node.line,
    line_end: node.line_end,
    section_lines: node.section_lines,
  };
  if (node.numbering !== null) {
    result.numbering = node.numbering;
  }
  if (node.is_likely_artifact === true) {
    result.is_likely_artifact = true;
    if (node.artifact_reason !== undefined) {
      result.artifact_reason = node.artifact_reason;
    }
  }
  if (node.children.length > 0) {
    result.children = node.children.map(compactTocNode);
  }
  return result;
}
```

   - else: subtree = index.toc.
3. Iterative depth reduction:
   ```typescript
   const requested = input.depth ?? 6;
   let trimmed: CompactTocNode[];
   let effectiveDepth = requested;
   let response: ViewTocResponse;
   for (let d = requested; d >= 1; d--) {
     trimmed = trimAndCompact(subtree, d);
     response = buildResponseObject(index, trimmed);
     const bytes = Buffer.byteLength(JSON.stringify(response), "utf8");
     if (bytes <= MAX_VIEW_TOC_BYTES) {
       effectiveDepth = d;
       if (d < requested) {
         response.truncated = true;
         response.effective_depth = d;
         response.hint = `Tree trimmed from depth ${requested} to ${d}; use start_id=<a leaf id> to drill deeper.`;
       }
       return response;
     }
   }
   // even depth=1 too big — take prefix
   const fallback = takePrefixWithinCap(subtree, MAX_VIEW_TOC_BYTES, index);
   return fallback;
   ```

```typescript
import { describe, it, expect, afterAll } from "vitest";
import { resolve, dirname } from "node:path";
import { fileURLToPath } from "node:url";
import { Client } from "@modelcontextprotocol/sdk/client/index.js";
import { InMemoryTransport } from "@modelcontextprotocol/sdk/inMemory.js";
import { createServer } from "../../src/server.js";
import { IndexCache } from "../../src/index/cache.js";

const __dirname = dirname(fileURLToPath(import.meta.url));
const TRM = resolve(__dirname, "../fixtures/public/esp32-p4-trm.md");

async function makeClient() {
  const [ct, st] = InMemoryTransport.createLinkedPair();
  const server = createServer({ cache: new IndexCache() });
  const client = new Client({ name: "stress-test", version: "0" });
  await Promise.all([server.connect(st), client.connect(ct)]);
  return client;
}

function parse(result: unknown) {
  const r = result as { content: Array<{ type: string; text: string }> };
  return JSON.parse(r.content[0]!.text);
}

describe("stress: huge document (ESP32-P4 TRM)", () => {
  const openClients: Array<{ close: () => Promise<void> }> = [];
  afterAll(async () => {
    for (const c of openClients) await c.close();
  });

  it("view_toc fits under cap with truncated=true on the full TRM", async () => {
    const client = await makeClient();
    openClients.push({ close: () => client.close() });
    const result = await client.callTool({ name: "view_toc", arguments: { file_path: TRM } });
    const parsed = parse(result);
    const jsonBytes = Buffer.byteLength(JSON.stringify(parsed), "utf8");
    expect(jsonBytes).toBeLessThanOrEqual(25 * 1024);
    expect(parsed.truncated).toBe(true);
    expect(parsed.effective_depth).toBeDefined();
    expect(parsed.hint).toMatch(/start_id|depth/i);
  }, 30_000);

  it("view_toc(start_id=<root child>) returns the subtree", async () => {
    const client = await makeClient();
    openClients.push({ close: () => client.close() });
    const root = parse(
      await client.callTool({ name: "view_toc", arguments: { file_path: TRM } })
    );
    const firstId = root.toc[0]?.id;
    expect(firstId).toBeTruthy();
    const sub = parse(
      await client.callTool({
        name: "view_toc",
        arguments: { file_path: TRM, start_id: firstId },
      })
    );
    expect(sub.toc).toBeDefined();
  }, 30_000);

  it("consecutive_pair_header detected on TRM (>= 30 cases)", async () => {
    const client = await makeClient();
    openClients.push({ close: () => client.close() });
    const result = parse(
      await client.callTool({ name: "analyze_document", arguments: { file_path: TRM } })
    );
    const pairs = result.anomalies.filter(
      (a: { type: string }) => a.type === "consecutive_pair_header"
    );
    expect(pairs.length).toBeGreaterThanOrEqual(30);
  }, 30_000);

  it("line-coverage invariant holds on TRM", async () => {
    const cache = new IndexCache();
    const idx = await cache.getOrBuild(TRM);
    // Same logic as tests/integration/invariants.test.ts but applied to TRM
    const owner: Array<string | undefined> = new Array(idx.line_count + 1);
    const firstHeading = idx.flat_headers[0];
    const preambleEnd = firstHeading ? firstHeading.line - 1 : idx.line_count;
    for (let l = 1; l <= preambleEnd; l++) owner[l] = "preamble";

    function visit(node: typeof idx.toc[number]) {
      const firstChild = node.children[0];
      const ownEnd = firstChild ? firstChild.line - 1 : node.line_end;
      for (let l = node.line; l <= ownEnd; l++) {
        expect(owner[l]).toBeUndefined();
        owner[l] = node.id;
      }
      for (const c of node.children) visit(c);
    }
    idx.toc.forEach(visit);

    for (let l = 1; l <= idx.line_count; l++) {
      expect(owner[l]).toBeDefined();
    }
  }, 30_000);
```
